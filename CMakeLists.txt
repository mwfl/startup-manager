cmake_minimum_required(VERSION 3.21)
project(startup_manager VERSION 0.1.0 LANGUAGES CXX)

if(POLICY CMP0169)
  cmake_policy(SET CMP0169 OLD)
endif()

include(FetchContent)
option(STARTUP_MANAGER_USE_LOCAL_MWFL "Use a local mwfl checkout" OFF)
set(STARTUP_MANAGER_MWFL_SOURCE_DIR "../mwfl" CACHE PATH "Local mwfl source directory")
set(MWFL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MWFL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(STARTUP_MANAGER_USE_LOCAL_MWFL)
  add_subdirectory("${STARTUP_MANAGER_MWFL_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/mwfl" EXCLUDE_FROM_ALL)
else()
  FetchContent_Declare(mwfl GIT_REPOSITORY https://github.com/mwfl/mwfl.git GIT_TAG v0.1.0)
  FetchContent_GetProperties(mwfl)
  if(NOT mwfl_POPULATED)
    FetchContent_Populate(mwfl)
    add_subdirectory("${mwfl_SOURCE_DIR}" "${mwfl_BINARY_DIR}" EXCLUDE_FROM_ALL)
  endif()
endif()
add_library(startup-manager-core src/startup_manager.cpp src/startup_manager.h src/startup_entry.h)
target_include_directories(startup-manager-core PUBLIC src)
target_link_libraries(startup-manager-core PUBLIC mwfl::security mwfl::deployment shell32 ole32 taskschd)
target_compile_features(startup-manager-core PUBLIC cxx_std_20)

add_executable(startup-manager WIN32 src/main.cpp src/startup-manager.rc src/resource.h
  app-icon.ico app.manifest)
target_link_libraries(startup-manager PRIVATE startup-manager-core mwfl::app shell32 advapi32)
target_compile_features(startup-manager PRIVATE cxx_std_20)
target_compile_definitions(startup-manager PRIVATE MWFL_APP_VERSION=L\"${PROJECT_VERSION}\")

enable_testing()
add_executable(startup-manager-tests tests/startup_entry_test.cpp)
target_link_libraries(startup-manager-tests PRIVATE startup-manager-core)
target_compile_definitions(startup-manager-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
foreach(target startup-manager-core startup-manager startup-manager-tests)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  endif()
endforeach()
add_test(NAME startup-manager.domain COMMAND startup-manager-tests)

install(TARGETS startup-manager RUNTIME DESTINATION . COMPONENT Application)
install(FILES README.md LICENSE CHANGELOG.md SECURITY.md DESTINATION . COMPONENT Application)
